- Maintains backwards compatibility: existing eventIds in KV remain valid (different format, but parsed the same way)

import { describe, it, expect } from "vitest";
import { SELF } from "cloudflare:test";

describe("Security Headers", () => {
  const requiredHeaders = [
    "Content-Security-Policy",
    "Strict-Transport-Security", 
    "Referrer-Policy",
    "Permissions-Policy",
    "X-Content-Type-Options",
    "X-Frame-Options",
  ];

  describe("on API responses", () => {
    it("includes all security headers on /api/health", async () => {
      const response = await SELF.fetch("http://localhost/api/health", {
        method: "GET",
      });
      
      for (const header of requiredHeaders) {
        expect(response.headers.has(header), `Missing header: ${header}`).toBe(true);
      }
    });

    it("includes all security headers on /api/events", async () => {
      const response = await SELF.fetch("http://localhost/api/events", {
        method: "GET",
      });
      
      for (const header of requiredHeaders) {
        expect(response.headers.has(header), `Missing header: ${header}`).toBe(true);
      }
    });

    it("includes all security headers on /webhook", async () => {
      const validPayload = {
        name: "Started",
        event: "com.jamf.setupmanager.started",
        timestamp: "2025-01-01T00:00:00Z",
        started: "2025-01-01T00:00:00Z",
        modelName: "MacBook Pro",
        modelIdentifier: "Mac15,3",
        macOSBuild: "24A335",
        macOSVersion: "15.0",
        serialNumber: "TEST001",
        setupManagerVersion: "2.0.0",
      };

      const response = await SELF.fetch("http://localhost/webhook", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(validPayload),
      });
      
      for (const header of requiredHeaders) {
        expect(response.headers.has(header), `Missing header: ${header}`).toBe(true);
      }
    });
  });

  describe("Strict-Transport-Security (SEC-02, D-03)", () => {
    it("has max-age of 31536000 (1 year)", async () => {
      const response = await SELF.fetch("http://localhost/api/health");
      const hsts = response.headers.get("Strict-Transport-Security");

      const response = await SELF.fetch("http://localhost/api/health");

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser -> Worker | Security headers protect against XSS, clickjacking, MIME sniffing |
| Attacker -> KV | UUID in event ID prevents key collision attacks |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Spoofing | Event ID | mitigate | crypto.randomUUID() appended per SEC-05 |
| T-03-02 | Tampering | Response headers | mitigate | CSP prevents XSS; X-Frame-Options/frame-ancestors prevent clickjacking |
| T-03-03 | Information Disclosure | Referrer | mitigate | strict-origin-when-cross-origin limits referrer leakage |
| T-03-04 | Denial of Service | Browser features | accept | Permissions-Policy disables unused features but doesn't prevent DoS |
| T-03-05 | Tampering | CSP unsafe-inline | accept | Required for Tailwind CSS v4 runtime styles; can be tightened with nonces in future |
</threat_model>
